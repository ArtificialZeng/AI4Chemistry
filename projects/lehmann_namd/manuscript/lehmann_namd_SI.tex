\subsection{Theorem 1: Generalized Causal Hellmann--Feynman and Dyson Residue Identity}

Let the real-symmetric lifted Hamiltonian be partitioned over the physical subspace $\mathcal{P} = \{|0\rangle, |S_1\rangle, \dots, |S_{N_S}\rangle\}$ ($n_{\mathrm{phys}} = 1 + N_S$) and the spin-pure bi-exciton bath $\mathcal{Q} = \{|D_1\rangle, \dots, |D_{N_D}\rangle\}$:
\begin{equation}
  \Hext(\mathbf{R}) =
  \begin{pmatrix}
    \mathbf{H}_{\mathrm{stat}}(\mathbf{R}) & \mathbf{V}(\mathbf{R}) \\
    \mathbf{V}(\mathbf{R})^\top & \mathbf{D}(\mathbf{R})
  \end{pmatrix}, \qquad
  \Hext(\mathbf{R})\,\mathbf{u}_n(\mathbf{R}) = E_n(\mathbf{R})\,\mathbf{u}_n(\mathbf{R}),
\end{equation}
with $\|\mathbf{u}_n\|_2^2 = \|\mathbf{c}_n\|_2^2 + \|\mathbf{b}_n\|_2^2 = 1$, where $\mathbf{u}_n = (\mathbf{c}_n^\top, \mathbf{b}_n^\top)^\top$.

\begin{proof}
From the $\mathcal{Q}$-block eigenvalue equation $\mathbf{V}^\top\mathbf{c}_n + \mathbf{D}\mathbf{b}_n = E_n\mathbf{b}_n$, whenever $E_n \notin \mathrm{spec}(\mathbf{D})$ we eliminate the bath amplitude analytically:
\begin{equation}
  \mathbf{b}_n = \mathbf{G}_n\,\mathbf{V}^\top\mathbf{c}_n, \qquad
  \mathbf{G}_n \equiv (E_n\mathbf{I} - \mathbf{D})^{-1}.
\end{equation}
Substituting into the $\mathcal{P}$-block yields the exact nonlinear frequency-dependent Dyson--Casida eigenproblem
\begin{equation}
  \bigl[\mathbf{H}_{\mathrm{stat}}(\mathbf{R}) + \mathbf{\Sigma}_{xc}(E_n; \mathbf{R})\bigr]\,\hat{\mathbf{c}}_n = E_n\,\hat{\mathbf{c}}_n, \qquad
  \mathbf{\Sigma}_{xc}(\omega; \mathbf{R}) = \mathbf{V}(\mathbf{R})\,(\omega\mathbf{I} - \mathbf{D}(\mathbf{R}))^{-1}\mathbf{V}(\mathbf{R})^\top,
\end{equation}
where $\hat{\mathbf{c}}_n = \mathbf{c}_n / \|\mathbf{c}_n\|_2$ is the unit-normalized physical component.
Using $\partial_\omega \mathbf{\Sigma}_{xc}(\omega; \mathbf{R}) = -\mathbf{V}(\omega\mathbf{I}-\mathbf{D})^{-2}\mathbf{V}^\top$, the normalization condition $\|\mathbf{c}_n\|^2 + \|\mathbf{b}_n\|^2 = 1$ becomes
\begin{equation}
  1 = \|\mathbf{c}_n\|^2 \Bigl(1 + \hat{\mathbf{c}}_n^\top \mathbf{V}\mathbf{G}_n^2\mathbf{V}^\top\hat{\mathbf{c}}_n\Bigr)
    = \|\mathbf{c}_n\|^2 \Bigl(1 - \hat{\mathbf{c}}_n^\top \bigl.\partial_\omega\mathbf{\Sigma}_{xc}\bigr|_{E_n}\hat{\mathbf{c}}_n\Bigr)
  \;\Longrightarrow\;
  Z_n \equiv \|\mathbf{c}_n\|^2 = \Bigl(1 - \hat{\mathbf{c}}_n^\top \bigl.\partial_\omega\mathbf{\Sigma}_{xc}\bigr|_{E_n}\hat{\mathbf{c}}_n\Bigr)^{-1}.
\end{equation}
Differentiating the static lifted eigenvalue identity $E_n = \mathbf{u}_n^\top\Hext\mathbf{u}_n$ with respect to nuclear coordinate $R_\alpha$ and applying $\mathbf{u}_n^\top(\nabla\mathbf{u}_n)=0$ yields
\begin{align}
  \nabla_{\mathbf{R}} E_n
  &= \mathbf{u}_n^\top (\nabla_{\mathbf{R}}\Hext)\mathbf{u}_n
   = \mathbf{c}_n^\top (\nabla_{\mathbf{R}}\mathbf{H}_{\mathrm{stat}})\mathbf{c}_n
   + \mathbf{c}_n^\top (\nabla_{\mathbf{R}}\mathbf{V})\mathbf{b}_n
   + \mathbf{b}_n^\top (\nabla_{\mathbf{R}}\mathbf{V}^\top)\mathbf{c}_n
   + \mathbf{b}_n^\top (\nabla_{\mathbf{R}}\mathbf{D})\mathbf{b}_n \nonumber \\
  &= Z_n\,\hat{\mathbf{c}}_n^\top \Bigl[
       \nabla_{\mathbf{R}}\mathbf{H}_{\mathrm{stat}}
     + (\nabla_{\mathbf{R}}\mathbf{V})\mathbf{G}_n\mathbf{V}^\top
     + \mathbf{V}\mathbf{G}_n(\nabla_{\mathbf{R}}\mathbf{V}^\top)
     + \mathbf{V}\mathbf{G}_n(\nabla_{\mathbf{R}}\mathbf{D})\mathbf{G}_n\mathbf{V}^\top
     \Bigr]\hat{\mathbf{c}}_n \nonumber \\
  &= Z_n\,\hat{\mathbf{c}}_n^\top \Bigl[\nabla_{\mathbf{R}}\mathbf{H}_{\mathrm{stat}}
     + \bigl(\nabla_{\mathbf{R}}\mathbf{\Sigma}_{xc}(\omega; \mathbf{R})\bigr)\big|_{\omega=E_n}\Bigr]\hat{\mathbf{c}}_n.
\end{align}
Thus the non-trivial frequency-derivative Dyson renormalization factor $Z_n \in (0, 1]$ is absorbed identically into static lifted expectation values, avoiding iterative nonlinear self-consistency in gradient evaluation.
Similarly, differentiating $\mathbf{u}_m^\top\Hext\mathbf{u}_n = 0$ ($m\ne n$) gives the exact non-adiabatic derivative coupling vector
\begin{equation}
  \mathbf{d}_{mn}^{\mathrm{CI}}(\mathbf{R})
  \equiv \mathbf{u}_m^\top (\nabla_{\mathbf{R}}\mathbf{u}_n)
  = \frac{\mathbf{h}_{mn}(\mathbf{R})}{E_n(\mathbf{R}) - E_m(\mathbf{R})}, \qquad
  \mathbf{h}_{mn}(\mathbf{R}) \equiv \mathbf{u}_m^\top (\nabla_{\mathbf{R}}\Hext(\mathbf{R}))\mathbf{u}_n.
\end{equation}
\end{proof}
